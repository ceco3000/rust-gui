129|129|
130|130|### 3.1 路由架构
131|131|
132|132|```rust
133|133|pub struct EventRouter {
134|134|    tree: WidgetTree,
135|135|    focus: FocusManager,
136|136|}
137|137|
138|138|pub struct WidgetTree {
139|139|    parent: FxHashMap<WidgetId, WidgetId>,
140|140|    children: FxHashMap<WidgetId, Vec<WidgetId>>,
141|141|    bounds: FxHashMap<WidgetId, Rect>,
142|142|}
143|143|
144|144|impl EventRouter {
145|145|    /// 分发事件到 widget 树。
146|146|    pub fn dispatch(
147|147|        &self, event: &Event,
148|148|        state_store: &mut StateStore, registry: &WidgetRegistry,
149|149|    ) {
150|150|        match event {
151|151|            Event::MouseDown { position, .. }
152|152|            | Event::MouseUp { position, .. }
153|153|            | Event::MouseMove { position, .. } => {
154|154|                if let Some(target) = self.hit_test(*position) {
155|155|                    self.dispatch_to_path(event, target, state_store, registry);
156|156|                }
157|157|            }
158|158|            Event::KeyDown { .. } | Event::KeyUp { .. } => {
159|159|                if let Some(id) = self.focus.current() {
160|160|                    self.dispatch_direct(event, id, state_store, registry);
161|161|                }
162|162|            }
163|163|            _ => {}
164|164|        }
165|165|    }
166|166|
167|167|    /// 沿树路径分发：捕获 → 目标 → 冒泡。
168|168|    fn dispatch_to_path(
169|169|        &self, event: &Event, target: WidgetId,
170|170|        state_store: &mut StateStore, registry: &WidgetRegistry,
171|171|    ) {
172|172|        let path = self.tree.path_to_root(target);
173|173|
174|174|        // 捕获阶段（从根向下，不包括 target）
175|175|        for &ancestor in path.iter().rev().skip(1) {
176|176|            if self.send_event(event, ancestor, EventPhase::Capture, state_store, registry) {
177|177|                return;
178|178|            }
179|179|        }
180|180|
181|181|        // 目标阶段
182|182|        if self.send_event(event, target, EventPhase::Target, state_store, registry) {
183|183|            return;
184|184|        }
185|185|
186|186|        // 冒泡阶段（从 target 向上，不包括根）
187|187|        for &ancestor in path.iter().skip(1) {
188|188|            if self.send_event(event, ancestor, EventPhase::Bubble, state_store, registry) {
189|189|                return;
190|190|            }
191|191|        }
192|192|    }
193|193|}
194|194|```

221|221|## 4. 命中测试
222|222|
223|223|```rust
224|224|impl WidgetTree {
225|225|    /// 命中测试：返回坐标处最深的 widget ID。
226|226|    ///
227|227|    /// 算法：
228|228|    /// 1. 从根节点开始
229|229|    /// 2. 检查当前节点包含坐标
230|230|    /// 3. 子节点按 z-index 降序排列后递归检查（高 z-index 优先命中，
231|231|    ///    无 z-index 时按 child 索引递增作为默认值，与 SceneGraph 的
232|232|    ///    z-order 排序一致）
233|233|    /// 4. 返回最深匹配
234|234|    pub fn hit_test(&self, point: Point) -> Option<WidgetId> {
235|235|        self.hit_test_recursive(self.root(), point)
236|236|    }
237|237|
238|238|    fn hit_test_recursive(&self, widget_id: WidgetId, point: Point) -> Option<WidgetId> {
239|239|        let bounds = self.bounds.get(&widget_id)?;
240|240|        if !bounds.contains(point) { return None; }
241|241|
242|242|        if let Some(children) = self.children.get(&widget_id) {
243|243|            // 按 z-index 降序排列子节点（z-order 与 SceneGraph 一致）
244|244|            let mut sorted: Vec<_> = children.iter().copied().collect();
245|245|            sorted.sort_by_key(|&id| {
246|246|                // 从 widget props 读取 z-index，无则用 0（与 scene_build 一致）
247|247|                self.z_index(id).unwrap_or(0)
248|248|            });
249|249|            for child in sorted.into_iter().rev() {
250|250|                if let Some(hit) = self.hit_test_recursive(child, point) {
251|251|                    return Some(hit);
252|252|                }
253|253|            }
254|254|        }
255|255|        Some(widget_id)
256|256|    }
257|257|
258|258|    fn z_index(&self, _widget_id: WidgetId) -> Option<i32> { /* 从 props 或 SceneGraph 读取 */ None }
259|259|
260|260|    pub fn root(&self) -> WidgetId { /* 返回 widget 树的根节点 ID */ todo!() }
261|261|
262|262|    pub fn traverse_visual_order(&self) -> Vec<WidgetId> { /* DFS 遍历产生视觉序列表 */ todo!() }
263|263|}
264|264|```
265|265|
266|266|**性能优化**：缓存最近命中测试结果（鼠标连续移动）；> 500 节点时使用 R-tree 空间索引；虚拟滚动容器只检查可见子节点。

455|455|## 9. 平台事件映射
456|456|
457|457|```rust
458|458|/// 将 winit 事件转换为框架 Event。
459|459|pub fn convert_winit_event(
460|460|    event: winit::event::WindowEvent,
461|461|    mouse_pos: Point,
462|462|    scale_factor: f64,
463|463|) -> Option<Event> {
464|464|    match event {
465|465|        winit::event::WindowEvent::MouseInput { state, button, .. } => {
466|466|            let btn = convert_button(button);
467|467|            let modifiers = Modifiers::current();
468|468|            match state {
469|469|                winit::event::ElementState::Pressed => {
470|470|                    Some(Event::MouseDown { position: mouse_pos, button: btn, modifiers })
471|471|                }
472|472|                winit::event::ElementState::Released => {
473|473|                    Some(Event::MouseUp { position: mouse_pos, button: btn, modifiers })
474|474|                }
475|475|            }
476|476|        }
477|477|        winit::event::WindowEvent::CursorMoved { position, .. } => {
478|478|            Some(Event::MouseMove {
479|479|                position: Point { x: position.x / scale_factor as f32, y: position.y / scale_factor as f32 },
480|480|                delta: Point { x: 0.0, y: 0.0 }, modifiers: Modifiers::current(),
481|481|            })
482|482|        }
483|483|        winit::event::WindowEvent::KeyboardInput { event, .. } => {
484|484|            let key = convert_key(event.physical_key);
485|485|            match event.state {
486|486|                winit::event::ElementState::Pressed => Some(Event::KeyDown {

### 10.1 事件队列

`App.events: Vec<Event>` 持有待处理的框架事件。handler 执行过程中可通过
`ctx.push_event()` 向队列追加事件（如 `request_redraw()` 推送 `MouseDown` 事件）。
队列在每帧 tick() 结束时 drain，所有事件在下一帧分发。

- 队列无容量上限——由调用方保证不高频堆积
- 滚动等高频场景需配合 throttle（≤ 1 次/帧）
